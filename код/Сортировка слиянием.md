```python
import random, time
import matplotlib.pyplot as plt

def merge_sort(A):
    if len(A) <= 1:
        return A
    
    mid = len(A) // 2
    left = merge_sort(A[:mid])
    right = merge_sort(A[mid:])
    
    return merge(left, right)

def merge(left, right):
    res = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] < right[j]:
            res.append(left[i]); i += 1
        else:
            res.append(right[j]); j += 1
    res.extend(left[i:]); res.extend(right[j:])
    return res

sizes = [10, 100, 500, 1000, 5000]
times = []

print("Прогрев: начат")
for _ in range(100):
    merge_sort([random.randint(0, 100) for _ in range(100)])
print("Прогрев: окончен")

for size in sizes:
    arr = [random.randint(0, 100) for _ in range(size)]
    avg_time = 0
    for _ in range(10):
        start = time.perf_counter()
        merge_sort(arr)
        avg_time += (time.perf_counter() - start)
    
    final_t = avg_time / 10
    times.append(final_t)
    print(f"Закончен подсчёт длины {size}:".ljust(40) + f"{final_t:.6f} сек.")

plt.plot(sizes, times, 'go-')
plt.title('Merge Sort Performance')
plt.xlabel('Размер данных (N)')
plt.ylabel('Время выполнения (T, сек)')
plt.grid(True)
plt.show()
```
