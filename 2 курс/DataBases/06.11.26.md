## Оператор GROUP BY
Группировка данных с одинаковыми значениями в указанных столбцах.
Пишется после оператора `WHERE` или `FROM`
Необязательный оператор, часто пишется с агрегатными функциями. 
## Оператор HAVING
Фильтрует значения. `WHERE` и `HAVING`Делают почти одну и ту же функцию. `HAVING` фильтрует после группировки. `HAVING` не может существовать без `GROW BY`.

# 7 задача
```SQL
SELECT type, COUNT(DISTINCT maker) AS maker_p
FROM (
	SELECT p.maker, p.type, pc.price
	FROM Productt p JOIN PC pc ON pc.model = p.model
	UNION ALL
	SELECT p.maker, p.type, l.price
	FROM Productt p JOIN Laptop l ON l.model = p.model
	UNION ALL
	SELECT p.maker, p.type, pr.price
	FROM Productt p JOIN Printer pr ON pr.model = p.model
) T
WHERE price > 400
GROUP BY type
ORDER BY type
```

### 7
```sql
DROP TABLE IF EXISTS Состав; 
DROP TABLE IF EXISTS Блюдо;
DROP TABLE IF EXISTS Кухня;
DROP TABLE IF EXISTS Ингредиент;
GO

CREATE TABLE Кухня (
    ID INT IDENTITY(1,1) NOT NULL, 
	Название NVARCHAR(70) NOT NULL,
    Описание NVARCHAR(100) NULL,
	
	CONSTRAINT  PK_Кухня_ID PRIMARY KEY (ID),
	CONSTRAINT UN_Кухня_Название UNIQUE (Название),
);

CREATE TABLE Блюдо (
	ID INT IDENTITY(1, 1) NOT NULL,
	Название NVARCHAR(70) NOT NULL,
	Цена MONEY NOT NULL,
	Вегетарианская BIT NOT NULL CONSTRAINT DE_Блюдо_Веген  DEFAULT 0,
	ID_кухня INT NULL,
	
	CONSTRAINT  PK_Блюдо_ID PRIMARY KEY (ID),
	CONSTRAINT CH_Блюдо_Цена CHECK (цена >= 0),
	CONSTRAINT  FK_Блюдо_ID_кухня FOREIGN KEY (ID_кухня) REFERENCES Кухня (ID)
);

CREATE TABLE Ингредиент (
	ID INT IDENTITY(1,1) NOT NULL, 
	Название NVARCHAR(70) NOT NULL,
	Единица_измерения NVARCHAR(10) NOT NULL,
	
	CONSTRAINT PK_Ингредиент_ID PRIMARY KEY (ID),
	CONSTRAINT UN_Ингредиент_Название UNIQUE (Название)
);

CREATE TABLE Состав (
    ID INT IDENTITY(1,1) NOT NULL,
    ID_блюда INT NULL,
    Кол_во_ингредиентов SMALLINT NOT NULL,
    ID_ингредиента INT NOT NULL,
    
	CONSTRAINT PK_Состав_ID PRIMARY KEY (ID),
    CONSTRAINT  FK_Состав_ID_блюда FOREIGN KEY (ID_блюда) REFERENCES Блюдо (ID),
    CONSTRAINT  FK_Состав_ID_ингредиента FOREIGN KEY (ID_ингредиента) REFERENCES Ингредиент (ID),
    CONSTRAINT UN_Состав_ID_блюда_ингредиента UNIQUE (ID_блюда, ID_ингредиента),
	CONSTRAINT CH_Состав_Кол_во_ингредиентов CHECK (Кол_во_ингредиентов >= 0)
);

SELECT * FROM Блюдо
SELECT * FROM Ингредиент
SELECT * FROM Кухня
SELECT * FROM Состав
```
