#Python #Информатика #Сортировка 

```python
import random, time
import matplotlib.pyplot as plt

def bubble_sort(A):
    A_copy = A.copy()
    n = len(A_copy)
    for i in range(n - 1):
        swapped = False
        for j in range(n - 1 - i):
            if A_copy[j] > A_copy[j + 1]:
                A_copy[j], A_copy[j + 1] = A_copy[j + 1], A_copy[j]
                swapped = True
        if not swapped:
            break
    return A_copy

sizes = [10, 100, 500, 1000, 5000]
times = []

print("Прогрев: начат")
for _ in range(100):
    bubble_sort([random.randint(0, 100) for _ in range(100)])
print("Прогрев: окончен")

for size in sizes:
    arr = [random.randint(0, 100) for _ in range(size)]
    avg_time = 0
    for _ in range(10):
        start = time.perf_counter()
        bubble_sort(arr)
        avg_time += (time.perf_counter() - start)
    
    final_t = avg_time / 10
    times.append(final_t)
    print(f"Закончен подсчёт длины {size}:".ljust(40) + f"{final_t:.6f} сек.")

plt.plot(sizes, times, 'ro-')
plt.title('Bubble Sort Performance')
plt.xlabel('Размер данных (N)')
plt.ylabel('Время выполнения (T, сек)')
plt.grid(True)
plt.show()
```
