## Основной интерфейс 
```python
import customtkinter as ctk
import pyperclip  # Для копирования в буфер обмена

# Настройка темы (Bitwarden-style: темно-синий и серый)
ctk.set_appearance_mode("Dark")
ctk.set_default_color_theme("dark-blue")

class PassManagerApp(ctk.CTk):
    def __init__(self):
        super().__init__()

        self.title("VaultGuard - Dashboard")
        self.geometry("900x600")

        # --- Сетка: 2 колонки ---
        # Колонка 0 (Список): узкая, не растягивается сильно
        # Колонка 1 (Детали): широкая, занимает всё оставшееся место
        self.grid_columnconfigure(0, weight=1)
        self.grid_columnconfigure(1, weight=3)
        self.grid_rowconfigure(0, weight=1)

        # === ЛЕВАЯ ПАНЕЛЬ (СПИСОК) ===
        self.sidebar_frame = ctk.CTkFrame(self, width=250, corner_radius=0)
        self.sidebar_frame.grid(row=0, column=0, sticky="nsew")
        self.sidebar_frame.grid_rowconfigure(2, weight=1) # Список растягивается

        # 1. Заголовок и Поиск
        self.logo_label = ctk.CTkLabel(self.sidebar_frame, text="Мой Сейф", font=ctk.CTkFont(size=20, weight="bold"))
        self.logo_label.grid(row=0, column=0, padx=20, pady=(20, 10))

        self.search_entry = ctk.CTkEntry(self.sidebar_frame, placeholder_text="Поиск...")
        self.search_entry.grid(row=1, column=0, padx=10, pady=10, sticky="ew")

        # 2. Скроллящийся список аккаунтов
        self.scroll_frame = ctk.CTkScrollableFrame(self.sidebar_frame, label_text="Аккаунты")
        self.scroll_frame.grid(row=2, column=0, padx=10, pady=10, sticky="nsew")

        # 3. Кнопка "Добавить" внизу
        self.add_button = ctk.CTkButton(self.sidebar_frame, text="+ Добавить запись", fg_color="green", hover_color="darkgreen")
        self.add_button.grid(row=3, column=0, padx=20, pady=20, sticky="ew")

        # === ПРАВАЯ ПАНЕЛЬ (ДЕТАЛИ) ===
        self.detail_frame = ctk.CTkFrame(self, corner_radius=0, fg_color="transparent")
        self.detail_frame.grid(row=0, column=1, sticky="nsew", padx=20, pady=20)
        
        # Заглушка, пока ничего не выбрано
        self.placeholder_label = ctk.CTkLabel(self.detail_frame, text="Выберите аккаунт слева", text_color="gray")
        self.placeholder_label.pack(expand=True)

        # --- ДЕМО ДАННЫЕ (Позже заменим на БД) ---
        self.mock_data = [
            {"service": "Google", "login": "my.email@gmail.com", "pass": "sUpErSecR3t!"},
            {"service": "VKontakte", "login": "id123456", "pass": "vk_pass_2025"},
            {"service": "Steam", "login": "gamer_pro", "pass": "GabeNewell123"},
            {"service": "Госуслуги", "login": "000-000-000 00", "pass": "PassportData!"},
            {"service": "GitHub", "login": "coder_python", "pass": "git_commit_push"},
        ]
        
        self.load_items_to_sidebar()

    def load_items_to_sidebar(self):
        # Очистка (если обновляем список)
        for widget in self.scroll_frame.winfo_children():
            widget.destroy()

        # Генерация кнопок для каждого аккаунта
        for item in self.mock_data:
            btn = ctk.CTkButton(
                self.scroll_frame, 
                text=item["service"], 
                image=None, # Сюда можно добавить иконки
                anchor="w", # Текст прижат влево
                fg_color="transparent", 
                text_color=("gray10", "gray90"),
                hover_color=("gray70", "gray30"),
                height=40,
                # Используем lambda, чтобы передать данные конкретного элемента
                command=lambda x=item: self.show_details(x)
            )
            btn.pack(fill="x", padx=5, pady=2)

    def show_details(self, item):
        # 1. Очищаем правую панель
        for widget in self.detail_frame.winfo_children():
            widget.destroy()

        # 2. Отрисовываем детали
        # Заголовок сервиса
        title = ctk.CTkLabel(self.detail_frame, text=item["service"], font=ctk.CTkFont(size=24, weight="bold"))
        title.pack(anchor="w", pady=(0, 20))

        # Поле Логин
        self.create_info_row("Логин / Email:", item["login"])

        # Поле Пароль
        self.create_info_row("Пароль:", item["pass"], is_password=True)

        # Кнопки действий (Редактировать / Удалить)
        action_frame = ctk.CTkFrame(self.detail_frame, fg_color="transparent")
        action_frame.pack(fill="x", pady=30)

        edit_btn = ctk.CTkButton(action_frame, text="Редактировать", width=100)
        edit_btn.pack(side="left", padx=(0, 10))
        
        del_btn = ctk.CTkButton(action_frame, text="Удалить", fg_color="red", hover_color="darkred", width=100)
        del_btn.pack(side="left")

    def create_info_row(self, label_text, value_text, is_password=False):
        """Вспомогательная функция для создания красивых строк с данными"""
        frame = ctk.CTkFrame(self.detail_frame, fg_color="transparent")
        frame.pack(fill="x", pady=5)

        lbl = ctk.CTkLabel(frame, text=label_text, text_color="gray", font=ctk.CTkFont(size=12))
        lbl.pack(anchor="w")

        # Контейнер для поля и кнопки копирования
        input_frame = ctk.CTkFrame(frame, fg_color="transparent")
        input_frame.pack(fill="x")

        # Если пароль - скрываем звездочками, иначе показываем текст
        show_char = "*" if is_password else ""
        
        entry = ctk.CTkEntry(input_frame, width=400)
        entry.insert(0, value_text)
        entry.configure(state="readonly", show=show_char) # Только чтение
        entry.pack(side="left", fill="x", expand=True)

        # Кнопка копирования
        copy_btn = ctk.CTkButton(
            input_frame, 
            text="Copy", 
            width=50, 
            fg_color="#444", 
            command=lambda: self.copy_to_clipboard(value_text)
        )
        copy_btn.pack(side="left", padx=(10, 0))

    def copy_to_clipboard(self, text):
        pyperclip.copy(text)
        print(f"Скопировано: {text}") # В реальности здесь можно вывести всплывающее уведомление

app = PassManagerApp()
app.mainloop()
```
